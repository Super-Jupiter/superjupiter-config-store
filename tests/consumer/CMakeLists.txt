cmake_minimum_required(VERSION 3.16)
project(SuperJupiterConfigStoreConsumer LANGUAGES CXX)
if(DEFINED SUPERJUPITER_CONFIG_STORE_SOURCE_DIR)
  add_subdirectory("${SUPERJUPITER_CONFIG_STORE_SOURCE_DIR}" config-store)
  if(TARGET superjupiter-config-store-tests OR TARGET superjupiter-config-store-replay)
    message(FATAL_ERROR "A consumer unexpectedly builds library developer tools")
  endif()
else()
  find_package(SuperJupiterConfigStore 0.1 CONFIG REQUIRED)
endif()
add_executable(superjupiter-config-store-consumer main.cpp)
target_link_libraries(superjupiter-config-store-consumer PRIVATE SuperJupiter::ConfigStore)
